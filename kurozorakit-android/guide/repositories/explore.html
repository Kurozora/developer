Explore Repository ​

The Explore Repository provides access to curated content discovery features.

Overview ​

kotlin
val exploreRepository = kurozoraKit.explore()
1

kotlin

Explore Categories:

  • trending: Currently trending anime
  • popular: Most popular anime
  • new_releases: Recently released shows
  • top_rated: Highest rated content
  • seasonal: Current season anime
  • upcoming: Upcoming releases
  • recommendations: Personalized recommendations

Example ​

kotlin
suspend fun loadExploreCategories() {
    val result = exploreRepository.getExploreCategories(limit = 10)
    
    when (result) {
        is Result.Success -> {
            result.data.data.forEach { category ->
                println("=== ${category.attributes.title} ===")
                println("${category.attributes.description}")
                println()
            }
        }
        is Result.Error -> handleError(result.error)
    }
}

// Load specific category content
suspend fun loadTrending() {
    val result = exploreRepository.getCategoryContent(
        categoryId = "trending",
        limit = 20
    )
    
    when (result) {
        is Result.Success -> {
            println("=== Trending Now ===")
            
            result.data.data.forEach { item ->
                when (item.type) {
                    "show" -> println("📺 ${item.attributes.title}")
                    "game" -> println("🎮 ${item.attributes.title}")
                    "literature" -> println("📚 ${item.attributes.title}")
                }
                println("  Rating: ${item.attributes.rating}/5")
                println()
            }
        }
        is Result.Error -> handleError(result.error)
    }
}

Explore UI:

kotlin
@Composable
fun ExploreScreen(viewModel: ExploreViewModel = viewModel()) {
    val categories by viewModel.categories.collectAsState()
    val isLoading by viewModel.isLoading.collectAsState()
    
    LaunchedEffect(Unit) {
        viewModel.loadCategories()
    }
    
    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("Explore") }
            )
        }
    ) { padding ->
        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding),
            contentPadding = PaddingValues(vertical = 16.dp)
        ) {
            items(categories, key = { it.id }) { category ->
                ExploreCategorySection(
                    category = category,
                    onSeeAll = { viewModel.navigateToCategory(category.id) }
                )
                
                Spacer(modifier = Modifier.height(24.dp))
            }
            
            if (isLoading) {
                item {
                    Box(
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(16.dp),
                        contentAlignment = Alignment.Center
                    ) {
                        CircularProgressIndicator()
                    }
                }
            }
        }
    }
}

@Composable
fun ExploreCategorySection(
    category: ExploreCategory,
    onSeeAll: () -> Unit
) {
    Column {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Column(modifier = Modifier.weight(1f)) {
                Text(
                    text = category.attributes.title,
                    style = MaterialTheme.typography.titleLarge,
                    fontWeight = FontWeight.Bold
                )
                
                category.attributes.description?.let { desc ->
                    Text(
                        text = desc,
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }
            }
            
            TextButton(onClick = onSeeAll) {
                Text("See All")
                Icon(
                    imageVector = Icons.Default.ChevronRight,
                    contentDescription = null
                )
            }
        }
        
        Spacer(modifier = Modifier.height(12.dp))
        
        LazyRow(
            contentPadding = PaddingValues(horizontal = 16.dp),
            horizontalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            items(category.content ?: emptyList(), key = { it.id }) { item ->
                ExploreCategoryItem(item = item)
            }
        }
    }
}

@Composable
fun ExploreCategoryItem(item: ExploreContent) {
    Card(
        modifier = Modifier
            .width(140.dp)
            .clickable { /* Navigate */ }
    ) {
        Column {
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(200.dp)
            ) {
                AsyncImage(
                    model = item.attributes.poster,
                    contentDescription = item.attributes.title,
                    modifier = Modifier.fillMaxSize(),
                    contentScale = ContentScale.Crop
                )
                
                // Type badge
                Surface(
                    modifier = Modifier
                        .align(Alignment.TopEnd)
                        .padding(8.dp),
                    color = when (item.type) {
                        "show" -> Color(0xFF2196F3)
                        "game" -> Color(0xFF4CAF50)
                        "literature" -> Color(0xFFFF9800)
                        else -> MaterialTheme.colorScheme.primary
                    },
                    shape = RoundedCornerShape(4.dp)
                ) {
                    Text(
                        text = item.type.uppercase(),
                        modifier = Modifier.padding(horizontal = 6.dp, vertical = 2.dp),
                        style = MaterialTheme.typography.labelSmall,
                        color = Color.White
                    )
                }
                
                // Rating
                item.attributes.rating?.let { rating ->
                    Surface(
                        modifier = Modifier
                            .align(Alignment.BottomEnd)
                            .padding(8.dp),
                        color = Color.Black.copy(alpha = 0.7f),
                        shape = RoundedCornerShape(4.dp)
                    ) {
                        Row(
                            modifier = Modifier.padding(horizontal = 6.dp, vertical = 4.dp),
                            verticalAlignment = Alignment.CenterVertically
                        ) {
                            Icon(
                                imageVector = Icons.Default.Star,
                                contentDescription = null,
                                tint = Color(0xFFFFC107),
                                modifier = Modifier.size(14.dp)
                            )
                            Spacer(modifier = Modifier.width(2.dp))
                            Text(
                                text = String.format("%.1f", rating),
                                style = MaterialTheme.typography.labelSmall,
                                color = Color.White
                            )
                        }
                    }
                }
            }
            
            Text(
                text = item.attributes.title,
                modifier = Modifier.padding(8.dp),
                style = MaterialTheme.typography.bodyMedium,
                maxLines = 2,
                overflow = TextOverflow.Ellipsis
            )
        }
    }
}
