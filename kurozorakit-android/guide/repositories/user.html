User Profile ​

kotlin
suspend fun loadCurrentUser() {
    val result = userRepository.getCurrentUser()
    
    when (result) {
        is Result.Success -> {
            val user = result.data.data.first()
            println("Username: ${user.attributes.username}")
            println("Email: ${user.attributes.email}")
            println("Member since: ${user.attributes.createdAt}")
            println("Bio: ${user.attributes.biography}")
        }
        is Result.Error -> {
            when (result.error) {
                is KurozoraError.AuthenticationError -> {
                    println("Not authenticated")
                }
                else -> handleError(result.error)
            }
        }
    }
}

// Update profile
suspend fun updateProfile(newUsername: String, newBio: String) {
    val result = userRepository.updateUserProfile(
        username = newUsername,
        biography = newBio
    )
    
    when (result) {
        is Result.Success -> {
            println("Profile updated successfully!")
        }
        is Result.Error -> handleError(result.error)
    }
}

// Load user library
suspend fun loadLibrary(status: LibraryStatus) {
    val result = userRepository.getUserLibrary(
        status = status,
        limit = 100
    )
    
    when (result) {
        is Result.Success -> {
            println("=== ${status.name} (${result.data.data.size}) ===")
            
            result.data.data.forEach { entry ->
                println("${entry.attributes.showTitle}")
                println("  Progress: ${entry.attributes.progress}/${entry.attributes.totalEpisodes}")
                entry.attributes.rating?.let { rating ->
                    println("  Rating: $rating/5")
                }
                println()
            }
        }
        is Result.Error -> handleError(result.error)
    }
}

// Load favorites
suspend fun loadFavorites(userId: String) {
    val result = userRepository.getUserFavorites(
        userId = userId,
        limit = 50
    )
    
    when (result) {
        is Result.Success -> {
            println("=== Favorites ===")
            
            val shows = result.data.shows
            val characters = result.data.characters
            val people = result.data.people
            
            if (shows.isNotEmpty()) {
                println("\nFavorite Shows:")
                shows.forEach { show ->
                    println("  📺 ${show.attributes.title}")
                }
            }
            
            if (characters.isNotEmpty()) {
                println("\nFavorite Characters:")
                characters.forEach { character ->
                    println("  👤 ${character.attributes.name}")
                }
            }
            
            if (people.isNotEmpty()) {
                println("\nFavorite People:")
                people.forEach { person ->
                    println("  🎭 ${person.attributes.fullName}")
                }
            }
        }
        is Result.Error -> handleError(result.error)
    }
}

class UserProfileViewModel(
    private val kurozoraKit: KurozoraKit
) : ViewModel() {
    
    private val _user = MutableStateFlow<User?>(null)
    val user: StateFlow<User?> = _user
    
    private val _stats = MutableStateFlow<UserStats?>(null)
    val stats: StateFlow<UserStats?> = _stats
    
    private val _library = MutableStateFlow<Map<LibraryStatus, List<LibraryEntry>>>(emptyMap())
    val library: StateFlow<Map<LibraryStatus, List<LibraryEntry>>> = _library
    
    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading
    
    fun loadUserProfile(userId: String? = null) {
        viewModelScope.launch {
            _isLoading.value = true
            
            // Load user profile
            val userResult = if (userId == null) {
                kurozoraKit.user().getCurrentUser()
            } else {
                kurozoraKit.user().getUser(userId)
            }
            
            if (userResult is Result.Success) {
                _user.value = userResult.data.data.first()
            }
            
            // Load user stats
            val statsResult = kurozoraKit.user().getUserStats(userId)
            if (statsResult is Result.Success) {
                _stats.value = statsResult.data.data.first()
            }
            
            // Load library
            loadLibrary(userId)
            
            _isLoading.value = false
        }
    }
    
    private suspend fun loadLibrary(userId: String?) {
        val libraryMap = mutableMapOf<LibraryStatus, List<LibraryEntry>>()
        
        LibraryStatus.values().forEach { status ->
            val result = kurozoraKit.user().getUserLibrary(
                userId = userId,
                status = status,
                limit = 100
            )
            
            if (result is Result.Success) {
                libraryMap[status] = result.data.data
            }
        }
        
        _library.value = libraryMap
    }
}

@Composable
fun UserProfileScreen(
    userId: String? = null,
    viewModel: UserProfileViewModel = viewModel()
) {
    val user by viewModel.user.collectAsState()
    val stats by viewModel.stats.collectAsState()
    val library by viewModel.library.collectAsState()
    val isLoading by viewModel.isLoading.collectAsState()
    
    LaunchedEffect(userId) {
        viewModel.loadUserProfile(userId)
    }
    
    if (isLoading || user == null) {
        Box(
            modifier = Modifier.fillMaxSize(),
            contentAlignment = Alignment.Center
        ) {
            CircularProgressIndicator()
        }
        return
    }
    
    LazyColumn(
        modifier = Modifier.fillMaxSize(),
        contentPadding = PaddingValues(16.dp),
        verticalArrangement = Arrangement.spacedBy(16.dp)
    ) {
        // Profile header
        item {
            UserProfileHeader(user = user!!)
        }
        
        // Stats
        if (stats != null) {
            item {
                UserStatsCard(stats = stats!!)
            }
        }
        
        // Library sections
        item {
            Text(
                text = "Anime Library",
                style = MaterialTheme.typography.titleLarge,
                fontWeight = FontWeight.Bold
            )
        }
        
        LibraryStatus.values().forEach { status ->
            val entries = library[status] ?: emptyList()
            
            if (entries.isNotEmpty()) {
                item {
                    LibrarySection(
                        status = status,
                        entries = entries
                    )
                }
            }
        }
    }
}

@Composable
fun UserProfileHeader(user: User) {
    Card(
        modifier = Modifier.fillMaxWidth()
    ) {
        Column(
            modifier = Modifier.padding(16.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            AsyncImage(
                model = user.attributes.profileImage,
                contentDescription = user.attributes.username,
                modifier = Modifier
                    .size(120.dp)
                    .clip(CircleShape),
                contentScale = ContentScale.Crop
            )
            
            Spacer(modifier = Modifier.height(12.dp))
            
            Text(
                text = user.attributes.username,
                style = MaterialTheme.typography.headlineSmall,
                fontWeight = FontWeight.Bold
            )
            
            user.attributes.biography?.let { bio ->
                Text(
                    text = bio,
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    textAlign = TextAlign.Center,
                    modifier = Modifier.padding(top = 8.dp)
                )
            }
            
            Spacer(modifier = Modifier.height(16.dp))
            
            Row(
                horizontalArrangement = Arrangement.spacedBy(24.dp)
            ) {
                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                    Text(
                        text = "${user.attributes.followersCount}",
                        style = MaterialTheme.typography.titleMedium,
                        fontWeight = FontWeight.Bold
                    )
                    Text(
                        text = "Followers",
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }
                
                Column(horizontalAlignment = Alignment.CenterHorizontally) {
                    Text(
                        text = "${user.attributes.followingCount}",
                        style = MaterialTheme.typography.titleMedium,
                        fontWeight = FontWeight.Bold
                    )
                    Text(
                        text = "Following",
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }
            }
            
            if (!user.attributes.isCurrentUser) {
                Spacer(modifier = Modifier.height(16.dp))
                
                Button(
                    onClick = { /* Follow/Unfollow */ },
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Text(if (user.attributes.isFollowing) "Unfollow" else "Follow")
                }
            }
        }
    }
}

@Composable
fun LibrarySection(
    status: LibraryStatus,
    entries: List<LibraryEntry>
) {
    Card(
        modifier = Modifier.fillMaxWidth()
    ) {
        Column(
            modifier = Modifier.padding(16.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = status.displayName,
                    style = MaterialTheme.typography.titleMedium,
                    fontWeight = FontWeight.Bold
                )
                
                Text(
                    text = "${entries.size}",
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.primary
                )
            }
            
            Spacer(modifier = Modifier.height(12.dp))
            
            LazyRow(
                horizontalArrangement = Arrangement.spacedBy(12.dp)
            ) {
                items(entries.take(10), key = { it.id }) { entry ->
                    LibraryEntryCard(entry = entry)
                }
                
                if (entries.size > 10) {
                    item {
                        SeeAllCard(count = entries.size - 10)
                    }
                }
            }
        }
    }
}

@Composable
fun LibraryEntryCard(entry: LibraryEntry) {
    Card(
        modifier = Modifier
            .width(120.dp)
            .clickable { /* Navigate */ }
    ) {
        Column {
            AsyncImage(
                model = entry.attributes.showPoster,
                contentDescription = entry.attributes.showTitle,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(180.dp),
                contentScale = ContentScale.Crop
            )
            
            Column(
                modifier = Modifier.padding(8.dp)
            ) {
                Text(
                    text = entry.attributes.showTitle,
                    style = MaterialTheme.typography.bodySmall,
                    maxLines = 2,
                    overflow = TextOverflow.Ellipsis
                )
                
                entry.attributes.rating?.let { rating ->
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier.padding(top = 4.dp)
                    ) {
                        Icon(
                            imageVector = Icons.Default.Star,
                            contentDescription = null,
                            tint = Color(0xFFFFC107),
                            modifier = Modifier.size(12.dp)
                        )
                        Spacer(modifier = Modifier.width(2.dp))
                        Text(
                            text = String.format("%.1f", rating),
                            style = MaterialTheme.typography.labelSmall
                        )
                    }
                }
            }
        }
    }
}
