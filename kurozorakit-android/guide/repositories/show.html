Show Repository ​

The Show Repository is the primary interface for accessing anime and TV show data from the Kurozora API. It provides comprehensive methods for browsing, searching, filtering, and managing shows along with their related data including seasons, episodes, characters, cast, staff, and more.

Overview ​

The Show Repository provides access to:

  • Anime series (TV, Movies, OVA, ONA, Special)
  • Detailed show information (synopsis, ratings, statistics)
  • Seasons and episodes
  • Characters and cast information
  • Production staff and studios
  • Theme songs
  • User reviews and ratings
  • Related shows and recommendations

Access the Show Repository through your KurozoraKit instance:

kotlin
val kurozoraKit = KurozoraKit.Builder()
    .apiKey("your-api-key")
    .platform(AndroidPlatform)
    .build()

val showRepository = kurozoraKit.show()

suspend fun loadShows() {
    val result = showRepository.getShows(limit = 20)
    
    when (result) {
        is Result.Success -> {
            val shows = result.data.data
            
            shows.forEach { show ->
                println("Title: ${show.attributes.title}")

                    println("API error: ${error.message}")
                }
                else -> {
                    println("Unknown error occurred")
                }
            }
        }
    }
}

Pagination Example:

kotlin
class ShowsViewModel : ViewModel() {
    
    private val _shows = MutableStateFlow<List<Show>>(emptyList())
    val shows: StateFlow<List<Show>> = _shows
    
    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading
    
    private var nextPageCursor: String? = null
    private var canLoadMore = true
    
    fun loadInitialShows() {
        viewModelScope.launch {
            _isLoading.value = true
            _shows.value = emptyList()
            nextPageCursor = null
            canLoadMore = true
            
            loadShows()
        }
    }
    
    fun loadMoreShows() {
        if (!canLoadMore || _isLoading.value) return
        
        viewModelScope.launch {
            loadShows()
        }
    }
    
    private suspend fun loadShows() {
        _isLoading.value = true
        
        val result = showRepository.getShows(
            next = nextPageCursor,
            limit = 20
        )
        
        when (result) {
            is Result.Success -> {
                val newShows = result.data.data
                _shows.value = _shows.value + newShows
                
                // Update pagination state
                nextPageCursor = result.data.links.next
                canLoadMore = nextPageCursor != null
            }
            
            is Result.Error -> {
                // Handle error
                _error.value = result.error.message
            }
        }
        
        _isLoading.value = false
    }
}

@Composable
fun ShowsListScreen(viewModel: ShowsViewModel = viewModel()) {
    val shows by viewModel.shows.collectAsState()
    val isLoading by viewModel.isLoading.collectAsState()
    
    LaunchedEffect(Unit) {
        viewModel.loadInitialShows()
    }
    
    LazyColumn(
        modifier = Modifier.fillMaxSize(),
        contentPadding = PaddingValues(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        items(shows, key = { it.id }) { show ->
            ShowCard(show = show)
        }
        
        // Load more when reaching end
        item {
            if (isLoading) {
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(16.dp),
                    contentAlignment = Alignment.Center
                ) {
                    CircularProgressIndicator()
                }
            }
        }
        
        // Trigger load more
        if (!isLoading && shows.isNotEmpty()) {
            item {
                LaunchedEffect(Unit) {
                    viewModel.loadMoreShows()
                }
            }
        }
    }
}

suspend fun loadShowEpisodes(showId: String) {
    var nextCursor: String? = null
    val allEpisodes = mutableListOf<Episode>()
    
    // Load all episodes with pagination
    do {
        val result = showRepository.getShowEpisodes(
            showId = showId,
            next = nextCursor,
            limit = 100
        )
        
        when (result) {
            is Result.Success -> {
                allEpisodes.addAll(result.data.data)
                nextCursor = result.data.links.next
            }
            is Result.Error -> {
                println("Error loading episodes: ${result.error.message}")
                break
            }
        }
    } while (nextCursor != null)
    
    println("Loaded ${allEpisodes.size} episodes")
    println()
    
    // Display episodes
    allEpisodes.forEach { episode ->
        val prefix = if (episode.attributes.isFiller) "[FILLER] " else ""
        val suffix = if (episode.attributes.isFinale) " [FINALE]" else ""
        
        println("$prefix Episode ${episode.attributes.numberTotal}: ${episode.attributes.title ?: "Untitled"}$suffix")
        println("  Duration: ${episode.attributes.duration} minutes")
        println("  Aired: ${episode.attributes.airedAt ?: "Not aired yet"}")
        episode.attributes.synopsis?.let { synopsis ->
            println("  Synopsis: ${synopsis.take(100)}...")
        }
        println()

suspend fun submitRating(showId: String, rating: Double, reviewText: String?) {
    // Validate rating
    if (rating < 0.0 || rating > 5.0) {
        println("Error: Rating must be between 0.0 and 5.0")
        return
    }
    
    // Validate review length
    if (reviewText != null && reviewText.length > 5000) {
        println("Error: Review must be less than 5000 characters")
        return
    }
    
    val result = showRepository.rateShow(
        showId = showId,
        rating = rating,
        review = reviewText
    )
    
    when (result) {
        is Result.Success -> {
            println("Rating submitted successfully!")
        }
        
        is Result.Error -> {
            when (result.error) {
                is KurozoraError.AuthenticationError -> {
                    println("You must be logged in to rate shows")
                }
                is KurozoraError.ValidationError -> {
                    println("Invalid rating: ${result.error.message}")
                }
                else -> {
                    println("Failed to submit rating: ${result.error.message}")
                }
            }
        }
    }
}

class ShowsBrowserViewModel(
    private val kurozoraKit: KurozoraKit
) : ViewModel() {
    
    private val _shows = MutableStateFlow<List<Show>>(emptyList())
    val shows: StateFlow<List<Show>> = _shows
    
    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading
    
    private val _error = MutableStateFlow<String?>(null)
    val error: StateFlow<String?> = _error
    
    private val _currentFilter = MutableStateFlow<ShowFilter?>(null)
    val currentFilter: StateFlow<ShowFilter?> = _currentFilter
    
    private var nextCursor: String? = null
    private var canLoadMore = true
    
    init {
        loadShows()
    }
    
    fun loadShows() {
        if (_isLoading.value) return
        
        viewModelScope.launch {
            _isLoading.value = true
            _error.value = null
            
            val result = kurozoraKit.show().getShows(
                next = null,
                limit = 20,
                filter = _currentFilter.value
            )
            
            when (result) {
                is Result.Success -> {
                    _shows.value = result.data.data
                    nextCursor = result.data.links.next
                    canLoadMore = nextCursor != null
                }
                
                is Result.Error -> {
                    _error.value = result.error.message ?: "Failed to load shows"
                }
            }
            
            _isLoading.value = false
        }
    }
    
    fun loadMore() {
        if (!canLoadMore || _isLoading.value) return
        
        viewModelScope.launch {
            _isLoading.value = true
            
            val result = kurozoraKit.show().getShows(
                next = nextCursor,
                limit = 20,
                filter = _currentFilter.value
            )
            
            when (result) {
                is Result.Success -> {
                    _shows.value = _shows.value + result.data.data
                    nextCursor = result.data.links.next
                    canLoadMore = nextCursor != null
                }
                
                is Result.Error -> {
                    _error.value = result.error.message
                }
            }
            
            _isLoading.value = false
        }
    }
    
    fun applyFilter(filter: ShowFilter) {
        _currentFilter.value = filter
        _shows.value = emptyList()
        nextCursor = null
        canLoadMore = true
        loadShows()
    }
    
    fun clearFilter() {
        _currentFilter.value = null
        _shows.value = emptyList()
        nextCursor = null
        canLoadMore = true
        loadShows()
    }
    
    fun refresh() {
        _shows.value = emptyList()
        nextCursor = null
        canLoadMore = true
        loadShows()
    }
}

fun ShowsBrowserScreen(
    viewModel: ShowsBrowserViewModel = viewModel(),
    onShowClick: (String) -> Unit
) {
    val shows by viewModel.shows.collectAsState()
    val isLoading by viewModel.isLoading.collectAsState()
    val error by viewModel.error.collectAsState()
    val currentFilter by viewModel.currentFilter.collectAsState()
    
    var showFilterSheet by remember { mutableStateOf(false) }
    
    val pullRefreshState = rememberPullRefreshState(
        refreshing = isLoading && shows.isEmpty(),
        onRefresh = { viewModel.refresh() }
    )

        topBar = {
            TopAppBar(
                title = { Text("Browse Anime") },

                    IconButton(onClick = { showFilterSheet = true }) {
                        Badge(
                            visible = currentFilter != null
                        ) {
                            Icon(Icons.Default.FilterList, "Filter")
                        }
                    }
                }
            )
        }
    ) { padding ->
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .pullRefresh(pullRefreshState)
        ) {
            when {
                error != null && shows.isEmpty() -> {
                    ErrorView(
                        message = error!!,
                        onRetry = { viewModel.loadShows() }
                    )
                }
                
                shows.isEmpty() && isLoading -> {
                    LoadingView()
                }
                
                shows.isEmpty() -> {
                    EmptyView(message = "No shows found")
                }
                
                else -> {
                    val listState = rememberLazyListState()

                        columns = GridCells.Adaptive(160.dp),

                        horizontalArrangement = Arrangement.spacedBy(12.dp),
                        verticalArrangement = Arrangement.spacedBy(12.dp)

                            ShowGridCard(

                                Box(
                                    modifier = Modifier
                                        .fillMaxWidth()
                                        .padding(16.dp),
                                    contentAlignment = Alignment.Center
                                ) {
                                    CircularProgressIndicator()
                                }
                            }
                        }

                        snapshotFlow { listState.layoutInfo.visibleItemsInfo.lastOrNull()?.index }

            onApplyFilter = { filter ->

    }
