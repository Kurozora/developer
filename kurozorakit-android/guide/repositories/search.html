Search Repository ​

The Search Repository provides universal search across all content types (shows, games, literature, characters, people, etc.).

Searchable Types:

  • shows: Anime shows
  • games: Video games
  • literatures: Manga and light novels
  • characters: Anime/game characters
  • people: Voice actors, directors, staff
  • studios: Animation studios
  • songs: Opening/ending themes
  • episodes: Individual episodes

Universal Search ​

kotlin
suspend fun performUniversalSearch(query: String) {
    val result = searchRepository.search(
        query = query,
        limit = 50
    )
    
    when (result) {
        is Result.Success -> {
            val shows = result.data.shows
            val games = result.data.games
            val literatures = result.data.literatures
            val characters = result.data.characters
            val people = result.data.people
            
            println("=== Search Results for: $query ===")
            
            if (shows.isNotEmpty()) {
                println("\nShows (${shows.size}):")
                shows.take(5).forEach { show ->
                    println("  📺 ${show.attributes.title}")
                }
            }
            
            if (games.isNotEmpty()) {
                println("\nGames (${games.size}):")
                games.take(5).forEach { game ->
                    println("  🎮 ${game.attributes.title}")
                }
            }
            
            if (literatures.isNotEmpty()) {
                println("\nLiterature (${literatures.size}):")
                literatures.take(5).forEach { lit ->
                    println("  📚 ${lit.attributes.title}")
                }
            }
            
            if (characters.isNotEmpty()) {
                println("\nCharacters (${characters.size}):")
                characters.take(5).forEach { char ->
                    println("  👤 ${char.attributes.name}")
                }
            }
            
            if (people.isNotEmpty()) {
                println("\nPeople (${people.size}):")
                people.take(5).forEach { person ->
                    println("  🎭 ${person.attributes.fullName}")
                }
            }
        }
        
        is Result.Error -> handleError(result.error)
    }
}

// Search only anime shows
suspend fun searchAnime(query: String) {
    val result = searchRepository.searchShows(
        query = query,
        limit = 20
    )
    
    when (result) {
        is Result.Success -> {
            println("=== Anime Search: $query ===")
            
            result.data.data.forEach { show ->
                println("${show.attributes.title}")
                println("  Type: ${show.attributes.type}")
                println("  Rating: ${show.attributes.stats.ratingAverage ?: "N/A"}")
                println("  Status: ${show.attributes.status}")
                println()
            }
        }
        is Result.Error -> handleError(result.error)
    }
}

// Search with filters
suspend fun searchHighRatedAnime(query: String) {
    val filter = ShowFilter(
        rating = FilterValue(min = 4.0),
        type = FilterValue(include = listOf(ShowType.TV))
    )
    
    val result = searchRepository.searchShows(
        query = query,
        filter = filter,
        limit = 20
    )
    
    when (result) {
        is Result.Success -> {
            println("=== High-Rated Anime: $query ===")
            
            result.data.data.forEach { show ->
                println("${show.attributes.title} - ${show.attributes.stats.ratingAverage}/5")
            }
        }
        is Result.Error -> handleError(result.error)
    }
}

Complete Search UI ​

kotlin
class SearchViewModel(
    private val kurozoraKit: KurozoraKit
) : ViewModel() {
    
    private val _searchQuery = MutableStateFlow("")
    val searchQuery: StateFlow<String> = _searchQuery
    
    private val _searchResults = MutableStateFlow<SearchResponse?>(null)
    val searchResults: StateFlow<SearchResponse?> = _searchResults
    
    private val _isSearching = MutableStateFlow(false)
    val isSearching: StateFlow<Boolean> = _isSearching
    
    private val _selectedTab = MutableStateFlow(0)
    val selectedTab: StateFlow<Int> = _selectedTab
    
    private var searchJob: Job? = null
    
    fun onSearchQueryChange(query: String) {
        _searchQuery.value = query
        
        searchJob?.cancel()
        
        if (query.isBlank()) {
            _searchResults.value = null
            return
        }
        
        searchJob = viewModelScope.launch {
            delay(500) // Debounce
            performSearch(query)
        }
    }
    
    private suspend fun performSearch(query: String) {
        _isSearching.value = true
        
        val result = kurozoraKit.search().search(
            query = query,
            limit = 50
        )
        
        when (result) {
            is Result.Success -> {
                _searchResults.value = result.data
            }
            is Result.Error -> {
                println("Search error: ${result.error.message}")
            }
        }
        
        _isSearching.value = false
    }
    
    fun selectTab(index: Int) {
        _selectedTab.value = index
    }
}

@Composable
fun SearchScreen(viewModel: SearchViewModel = viewModel()) {
    val searchQuery by viewModel.searchQuery.collectAsState()
    val searchResults by viewModel.searchResults.collectAsState()
    val isSearching by viewModel.isSearching.collectAsState()
    val selectedTab by viewModel.selectedTab.collectAsState()
    
    Scaffold(
        topBar = {
            Column {
                TopAppBar(
                    title = {
                        SearchBar(
                            query = searchQuery,
                            onQueryChange = viewModel::onSearchQueryChange,
                            placeholder = "Search anime, games, characters..."
                        )
                    }
                )
                
                if (searchResults != null) {
                    ScrollableTabRow(
                        selectedTabIndex = selectedTab,
                        modifier = Modifier.fillMaxWidth()
                    ) {
                        Tab(
                            selected = selectedTab == 0,
                            onClick = { viewModel.selectTab(0) },
                            text = { Text("All (${getTotalCount(searchResults)})") }
                        )
                        Tab(
                            selected = selectedTab == 1,
                            onClick = { viewModel.selectTab(1) },
                            text = { Text("Shows (${searchResults?.shows?.size ?: 0})") }
                        )
                        Tab(
                            selected = selectedTab == 2,
                            onClick = { viewModel.selectTab(2) },
                            text = { Text("Games (${searchResults?.games?.size ?: 0})") }
                        )
                        Tab(
                            selected = selectedTab == 3,
                            onClick = { viewModel.selectTab(3) },
                            text = { Text("Literature (${searchResults?.literatures?.size ?: 0})") }
                        )
                        Tab(
                            selected = selectedTab == 4,
                            onClick = { viewModel.selectTab(4) },
                            text = { Text("Characters (${searchResults?.characters?.size ?: 0})") }
                        )
                        Tab(
                            selected = selectedTab == 5,
                            onClick = { viewModel.selectTab(5) },
                            text = { Text("People (${searchResults?.people?.size ?: 0})") }
                        )
                    }
                }
            }
        }
    ) { padding ->
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
        ) {
            when {
                isSearching -> {
                    Box(
                        modifier = Modifier.fillMaxSize(),
                        contentAlignment = Alignment.Center
                    ) {
                        CircularProgressIndicator()
                    }
                }
                
                searchQuery.isBlank() -> {
                    SearchEmptyState()
                }
                
                searchResults == null -> {
                    Box(
                        modifier = Modifier.fillMaxSize(),
                        contentAlignment = Alignment.Center
                    ) {
                        Text("No results found")
                    }
                }
                
                else -> {
                    SearchResultsList(
                        results = searchResults!!,
                        selectedTab = selectedTab
                    )
                }
            }
        }
    }
}

@Composable
fun SearchBar(
    query: String,
    onQueryChange: (String) -> Unit,
    placeholder: String
) {
    TextField(
        value = query,
        onValueChange = onQueryChange,
        modifier = Modifier.fillMaxWidth(),
        placeholder = { Text(placeholder) },
        leadingIcon = {
            Icon(Icons.Default.Search, contentDescription = "Search")
        },
        trailingIcon = {
            if (query.isNotEmpty()) {
                IconButton(onClick = { onQueryChange("") }) {
                    Icon(Icons.Default.Close, contentDescription = "Clear")
                }
            }
        },
        singleLine = true,
        colors = TextFieldDefaults.colors(
            focusedContainerColor = MaterialTheme.colorScheme.surface,
            unfocusedContainerColor = MaterialTheme.colorScheme.surface
        )
    )
}

@Composable
fun SearchResultsList(
    results: SearchResponse,
    selectedTab: Int
) {
    LazyColumn(
        modifier = Modifier.fillMaxSize(),
        contentPadding = PaddingValues(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        when (selectedTab) {
            0 -> { // All
                if (results.shows.isNotEmpty()) {
                    item {
                        Text(
                            text = "Shows",
                            style = MaterialTheme.typography.titleMedium,
                            fontWeight = FontWeight.Bold
                        )
                    }
                    items(results.shows.take(5)) { show ->
                        ShowSearchCard(show = show)
                    }
                }
                
                if (results.games.isNotEmpty()) {
                    item {
                        Spacer(modifier = Modifier.height(8.dp))
                        Text(
                            text = "Games",
                            style = MaterialTheme.typography.titleMedium,
                            fontWeight = FontWeight.Bold
                        )
                    }
                    items(results.games.take(5)) { game ->
                        GameSearchCard(game = game)
                    }
                }
                
                if (results.characters.isNotEmpty()) {
                    item {
                        Spacer(modifier = Modifier.height(8.dp))
                        Text(
                            text = "Characters",
                            style = MaterialTheme.typography.titleMedium,
                            fontWeight = FontWeight.Bold
                        )
                    }
                    items(results.characters.take(5)) { character ->
                        CharacterSearchCard(character = character)
                    }
                }
            }
            
            1 -> { // Shows
                items(results.shows) { show ->
                    ShowSearchCard(show = show)
                }
            }
            
            2 -> { // Games
                items(results.games) { game ->
                    GameSearchCard(game = game)
                }
            }
            
            3 -> { // Literature
                items(results.literatures) { literature ->
                    LiteratureSearchCard(literature = literature)
                }
            }
            
            4 -> { // Characters
                items(results.characters) { character ->
                    CharacterSearchCard(character = character)
                }
            }
            
            5 -> { // People
                items(results.people) { person ->
                    PersonSearchCard(person = person)
                }
            }
        }
    }
}

@Composable
fun ShowSearchCard(show: Show) {
    Card(
        modifier = Modifier
            .fillMaxWidth()
            .clickable { /* Navigate */ }
    ) {
        Row(
            modifier = Modifier.padding(12.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            AsyncImage(
                model = show.attributes.poster?.small,
                contentDescription = show.attributes.title,
                modifier = Modifier
                    .size(60.dp, 90.dp)
                    .clip(RoundedCornerShape(4.dp)),
                contentScale = ContentScale.Crop
            )
            
            Spacer(modifier = Modifier.width(12.dp))
            
            Column(modifier = Modifier.weight(1f)) {
                Text(
                    text = show.attributes.title,
                    style = MaterialTheme.typography.titleMedium,
                    fontWeight = FontWeight.Medium,
                    maxLines = 2,
                    overflow = TextOverflow.Ellipsis
                )
                
                Row(
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = show.attributes.type.toString(),
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.primary
                    )
                    
                    show.attributes.stats.ratingAverage?.let { rating ->
                        Text(" • ")
                        Icon(
                            imageVector = Icons.Default.Star,
                            contentDescription = null,
                            tint = Color(0xFFFFC107),
                            modifier = Modifier.size(14.dp)
                        )
                        Text(
                            text = String.format("%.1f", rating),
                            style = MaterialTheme.typography.bodySmall
                        )
                    }
                }
                
                show.attributes.synopsis?.let { synopsis ->
                    Text(
                        text = synopsis,
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                        maxLines = 2,
                        overflow = TextOverflow.Ellipsis,
                        modifier = Modifier.padding(top = 4.dp)
                    )
                }
            }
        }
    }
}

            )

            Text(
                text = "Try searching for your favorite shows or characters",
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.7f)
            )
        }
    }
}

fun getTotalCount(results: SearchResponse?): Int {
    if (results == null) return 0
    return results.shows.size + 
           results.games.size + 
           results.literatures.size + 
           results.characters.size + 
           results.people.size + 
           results.studios.size
}
