Feed Repository ​

The Feed Repository provides access to personalized activity feeds and user updates.

Overview ​

kotlin
val feedRepository = kurozoraKit.feed()
1

FeedMessage Types:

  • REVIEW: User posted a review
  • RATING: User rated content
  • LIBRARY_UPDATE: User updated their library
  • FOLLOW: User followed someone
  • WATCH_STATUS: User watched an episode
  • ACHIEVEMENT: User earned an achievement

Example ​

kotlin
suspend fun loadUserFeed() {
    val result = feedRepository.getFeed(limit = 50)
    
    when (result) {
        is Result.Success -> {
            result.data.data.forEach { message ->
                when (message.attributes.type) {
                    "REVIEW" -> {
                        println("${message.attributes.username} reviewed ${message.attributes.contentTitle}")
                        println("  ${message.attributes.review}")
                    }
                    "RATING" -> {
                        println("${message.attributes.username} rated ${message.attributes.contentTitle}: ${message.attributes.rating}/5")
                    }
                    "LIBRARY_UPDATE" -> {
                        println("${message.attributes.username} added ${message.attributes.contentTitle} to ${message.attributes.libraryStatus}")
                    }
                    "WATCH_STATUS" -> {
                        println("${message.attributes.username} watched ${message.attributes.contentTitle} - Episode ${message.attributes.episodeNumber}")
                    }
                    "FOLLOW" -> {
                        println("${message.attributes.username} started following ${message.attributes.targetUsername}")
                    }
                    else -> {
                        println("${message.attributes.username}: ${message.attributes.message}")
                    }
                }
                println("  ${message.attributes.createdAt}")
                println()
            }
        }
        is Result.Error -> handleError(result.error)
    }
}

// Load explore feed
suspend fun loadExploreFeed() {
    val result = feedRepository.getExploreFeed(limit = 30)
    
    when (result) {
        is Result.Success -> {
            println("=== Trending on Kurozora ===")
            
            result.data.data.forEach { message ->
                println("${message.attributes.contentTitle}")
                println("  ${message.attributes.message}")
                println("  ${message.attributes.likesCount} likes")
                println()
            }
        }
        is Result.Error -> handleError(result.error)
    }
}

Feed UI:

kotlin
@Composable
fun FeedScreen(viewModel: FeedViewModel = viewModel()) {
    val feed by viewModel.feed.collectAsState()
    val isLoading by viewModel.isLoading.collectAsState()
    
    LaunchedEffect(Unit) {
        viewModel.loadFeed()
    }
    
    val pullRefreshState = rememberPullRefreshState(
        refreshing = isLoading && feed.isEmpty(),
        onRefresh = { viewModel.refresh() }
    )
    
    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("Feed") }
            )
        }
    ) { padding ->
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .pullRefresh(pullRefreshState)
        ) {
            LazyColumn(
                modifier = Modifier.fillMaxSize(),
                contentPadding = PaddingValues(vertical = 8.dp)
            ) {
                items(feed, key = { it.id }) { message ->
                    FeedMessageCard(message = message)
                    Divider()
                }
                
                if (isLoading) {
                    item {
                        Box(
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(16.dp),
                            contentAlignment = Alignment.Center
                        ) {
                            CircularProgressIndicator()
                        }
                    }
                }
            }
            
            PullRefreshIndicator(
                refreshing = isLoading && feed.isEmpty(),
                state = pullRefreshState,
                modifier = Modifier.align(Alignment.TopCenter)
            )
        }
    }
}

@Composable
fun FeedMessageCard(message: FeedMessage) {
    Card(
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 16.dp, vertical = 8.dp),
        elevation = CardDefaults.cardElevation(defaultElevation = 2.dp)
    ) {
        Column(modifier = Modifier.padding(16.dp)) {
            // User info
            Row(
                verticalAlignment = Alignment.CenterVertically
            ) {
                AsyncImage(
                    model = message.attributes.userAvatar,
                    contentDescription = message.attributes.username,
                    modifier = Modifier
                        .size(40.dp)
                        .clip(CircleShape)
                )
                
                Spacer(modifier = Modifier.width(12.dp))
                
                Column(modifier = Modifier.weight(1f)) {
                    Text(
                        text = message.attributes.username,
                        style = MaterialTheme.typography.titleSmall,
                        fontWeight = FontWeight.Bold
                    )
                    
                    Text(
                        text = message.attributes.createdAt,
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }
                
                // Type badge
                Surface(
                    color = when (message.attributes.type) {
                        "REVIEW" -> Color(0xFF4CAF50)
                        "RATING" -> Color(0xFFFFC107)
                        "WATCH_STATUS" -> Color(0xFF2196F3)
                        else -> MaterialTheme.colorScheme.secondaryContainer
                    },
                    shape = RoundedCornerShape(4.dp)
                ) {
                    Text(
                        text = message.attributes.type,
                        modifier = Modifier.padding(horizontal = 8.dp, vertical = 4.dp),
                        style = MaterialTheme.typography.labelSmall,
                        color = Color.White
                    )
                }
            }
            
            Spacer(modifier = Modifier.height(12.dp))
            
            // Message content
            when (message.attributes.type) {
                "REVIEW" -> {
                    Text(
                        text = "Reviewed ${message.attributes.contentTitle}",
                        style = MaterialTheme.typography.bodyMedium,
                        fontWeight = FontWeight.Medium
                    )
                    
                    Spacer(modifier = Modifier.height(8.dp))
                    
                    message.attributes.rating?.let { rating ->
                        Row(verticalAlignment = Alignment.CenterVertically) {
                            repeat(5) { index ->
                                Icon(
                                    imageVector = if (index < rating.toInt()) {
                                        Icons.Default.Star
                                    } else {
                                        Icons.Default.StarBorder
                                    },
                                    contentDescription = null,
                                    tint = Color(0xFFFFC107),
                                    modifier = Modifier.size(16.dp)
                                )
                            }
                            Spacer(modifier = Modifier.width(8.dp))
                            Text(
                                text = String.format("%.1f", rating),
                                style = MaterialTheme.typography.bodyMedium
                            )
                        }
                    }
                    
                    message.attributes.review?.let { review ->
                        Spacer(modifier = Modifier.height(8.dp))
                        Text(
                            text = review,
                            style = MaterialTheme.typography.bodyMedium,
                            maxLines = 3,
                            overflow = TextOverflow.Ellipsis
                        )
                    }
                }
                
                "WATCH_STATUS" -> {
                    Text(
                        text = "Watched ${message.attributes.contentTitle}",
                        style = MaterialTheme.typography.bodyMedium,
                        fontWeight = FontWeight.Medium
                    )
                    
                    message.attributes.episodeNumber?.let { episode ->
                        Text(
                            text = "Episode $episode",
                            style = MaterialTheme.typography.bodySmall,
                            color = MaterialTheme.colorScheme.onSurfaceVariant
                        )
                    }
                }
                
                else -> {
                    Text(
                        text = message.attributes.message,
                        style = MaterialTheme.typography.bodyMedium
                    )
                }
            }
            
            // Actions
            Spacer(modifier = Modifier.height(12.dp))
            
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Row {
                    TextButton(onClick = { /* Like */ }) {
                        Icon(
                            imageVector = if (message.attributes.isLiked) {
                                Icons.Default.Favorite
                            } else {
                                Icons.Default.FavoriteBorder
                            },
                            contentDescription = "Like"
                        )
                        Spacer(modifier = Modifier.width(4.dp))
                        Text("${message.attributes.likesCount}")
                    }
                    
                    TextButton(onClick = { /* Comment */ }) {
                        Icon(Icons.Default.Comment, contentDescription = "Comment")
                        Spacer(modifier = Modifier.width(4.dp))
                        Text("${message.attributes.commentsCount}")
                    }
                }
                
                IconButton(onClick = { /* Share */ }) {
                    Icon(Icons.Default.Share, contentDescription = "Share")
                }
            }
        }
    }
}
