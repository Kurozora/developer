Literature Repository ​

The Literature Repository provides access to manga, light novels, and other literary works from the Kurozora API.

Overview ​

Access the Literature Repository through KurozoraKit:

kotlin
val literatureRepository = kurozoraKit.literature()
1

Fetch a paginated list of literary works:

kotlin
suspend fun getLiteratures(

kotlin
val result = kurozoraKit.literature().getLiteratures(limit = 20)

when (result) {
    is Result.Success -> {
        result.data.data.forEach { literature ->
            println("${literature.attributes.title} (${literature.attributes.type})")
            println("Volumes: ${literature.attributes.volumeCount}")
            println("Chapters: ${literature.attributes.chapterCount}")

}

Search ​

Universal search across multiple content types:

kotlin
suspend fun search(
    scope: KKSearchScope,
    types: List<KKSearchType>,
    query: String,
    next: String? = null,
    limit: Int = 20,
    filter: KKSearchFilter? = null
): Result<SearchResponse>
1
2
3
4
5
6
7
8

Parameters:

  • scope - Search scope:

  • KKSearchScope.kurozora - Search all public content

  • KKSearchScope.library - Search user's library only

  • types - Content types to search:

  • KKSearchType.shows - Anime/TV shows

  • KKSearchType.literatures - Manga/novels

  • KKSearchType.games - Video games

  • KKSearchType.characters - Characters

  • KKSearchType.people - People (staff, actors)

  • KKSearchType.studios - Studios

  • KKSearchType.songs - Songs

  • KKSearchType.episodes - Episodes

Example:

kotlin
// Search for anime shows
val result = kurozoraKit.search().search(
    scope = KKSearchScope.kurozora,
    types = listOf(KKSearchType.shows),
    query = "demon slayer",
    limit = 20
)

when (result) {
    is Result.Success -> {
        result.data.shows?.forEach { show ->
            println("${show.attributes.title}")
            println("Score: ${show.attributes.stats.ratingAverage}")
        }
    }
    is Result.Error -> handleError(result.error)
}

Example:

kotlin
// Get suggestions as user types
fun onSearchQueryChanged(query: String) {
    if (query.length >= 2) {
        viewModelScope.launch {
            val result = kurozoraKit.search().getSearchSuggestions(
                scope = KKSearchScope.kurozora,
                types = listOf(KKSearchType.shows),
                query = query
            )
            
            when (result) {
                is Result.Success -> {
                    _suggestions.value = result.data.suggestions
                }
                is Result.Error -> { /* ignore */ }
            }
        }
    }
}

Search UI Example ​

Complete search implementation with Compose:

kotlin
@Composable
fun SearchScreen(viewModel: SearchViewModel = viewModel()) {
    var query by remember { mutableStateOf("") }
    val searchResults by viewModel.searchResults.collectAsState()
    val suggestions by viewModel.suggestions.collectAsState()
    val isSearching by viewModel.isSearching.collectAsState()
    
    Column(modifier = Modifier.fillMaxSize()) {
        // Search bar
        OutlinedTextField(
            value = query,
            onValueChange = { 
                query = it
                viewModel.onSearchQueryChanged(it)
            },
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp),
            placeholder = { Text("Search anime, manga, games...") },
            leadingIcon = {
                Icon(Icons.Default.Search, contentDescription = null)
            },
            trailingIcon = {
                if (query.isNotEmpty()) {
                    IconButton(onClick = { 
                        query = ""
                        viewModel.clearSearch()
                    }) {
                        Icon(Icons.Default.Close, contentDescription = "Clear")
                    }
                }
            }
        )
        
        when {
            // Show suggestions while typing
            query.isNotEmpty() && suggestions.isNotEmpty() && searchResults.isEmpty() -> {
                LazyColumn {
                    items(suggestions) { suggestion ->
                        ListItem(
                            headlineContent = { Text(suggestion) },
                            modifier = Modifier.clickable {
                                query = suggestion
                                viewModel.search(suggestion)
                            }
                        )
                    }
                }
            }
            
            // Show search results
            searchResults.isNotEmpty() -> {
                LazyColumn {
                    // Shows section
                    searchResults.shows?.let { shows ->
                        if (shows.isNotEmpty()) {
                            item {
                                Text(
                                    "Shows",
                                    style = MaterialTheme.typography.titleMedium,
                                    modifier = Modifier.padding(16.dp)
                                )
                            }
                            items(shows) { show ->
                                ShowSearchResultCard(show)
                            }
                        }
                    }
                    
                    // Characters section
                    searchResults.characters?.let { characters ->
                        if (characters.isNotEmpty()) {
                            item {
                                Text(
                                    "Characters",
                                    style = MaterialTheme.typography.titleMedium,
                                    modifier = Modifier.padding(16.dp)
                                )
                            }
                            items(characters) { character ->
                                CharacterSearchResultCard(character)
                            }
                        }
                    }
                }
            }
            
            // Loading
            isSearching -> {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = Alignment.Center
                ) {
                    CircularProgressIndicator()
                }
            }
            
            // Empty state
            query.isEmpty() -> {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = Alignment.Center
                ) {
                    Column(horizontalAlignment = Alignment.CenterHorizontally) {
                        Icon(
                            Icons.Default.Search,
                            contentDescription = null,
                            modifier = Modifier.size(64.dp),
                            tint = MaterialTheme.colorScheme.onSurfaceVariant
                        )
                        Spacer(modifier = Modifier.height(16.dp))
                        Text("Search for anime, manga, and more")
                    }
                }
            }
        }
    }
}

Usage with Filters ​

Use genres and themes to filter content:

kotlin
// Get all genres first
val genresResult = kurozoraKit.genre().getGenres()
val actionGenreId = genresResult.getOrNull()
    ?.data
    ?.find { it.attributes.name == "Action" }
    ?.id

// Filter shows by genre
val showsResult = kurozoraKit.show().getShows(
    filter = ShowFilter(
        genres = FilterValue(include = actionGenreId)
    )
)
