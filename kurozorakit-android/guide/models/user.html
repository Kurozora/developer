User Model ​

Represents user accounts and profiles.

Structure ​

kotlin
@Serializable
data class User(
    override val id: String,
    val uuid: String,
    override val type: String,
    override val href: String,
    val attributes: Attributes,
    val relationships: Relationships? = null
) : IdentityResource

Companion Object ​

The User model includes a companion object for global user state:

kotlin
companion object {
    var current: User? = null
    
    val isSignedIn: Boolean
        get() = current != null
    
    val isPro: Boolean
        get() = current?.attributes?.isPro ?: false
    
    val isSubscribed: Boolean
        get() = current?.attributes?.isSubscribed ?: false
}

Status Management ​

The User model includes smart status handling:

kotlin
var blockStatus: BlockStatus
    get() = _blockStatus ?: BlockStatus.fromBoolean(isBlocked)
    set(value) { _blockStatus = value }

var followStatus: FollowStatus
    get() = _followStatus ?: FollowStatus.fromBoolean(isFollowed)
    set(value) { _followStatus = value }

Update Methods ​

kotlin
// Update from UserUpdate
fun update(using: UserUpdate) {
    slug = using.username.toString()
    username = using.nickname.toString()
    biography = using.biography
    profile = using.profile
    banner = using.banner
    // ... etc
}

// Update from BlockUpdate
fun update(using: BlockUpdate) {
    blockStatus = using.blockStatus
}

// Update from FollowUpdate
fun update(using: FollowUpdate) {
    followStatus = using.followStatus
}

// Update subscription status
fun updateSubscription(from: Receipt) {
    val valid = from.attributes.isValid
    isSubscribed = valid
    isPro = valid
}

Usage Examples ​

Accessing Current User ​

kotlin
// Check if signed in
if (User.isSignedIn) {
    val currentUser = User.current
    println("Welcome, ${currentUser?.attributes?.username}!")
    
    if (User.isPro) {
        println("You have Pro access!")
    }
}

    biography = "Updated bio",
    preferredLanguage = "en"
)

user.attributes.update(using = update)
