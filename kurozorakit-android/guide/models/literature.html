Literature Model ​

Represents manga, light novels, and other written publications.

Structure ​

kotlin
@Serializable
data class Literature(
    override val id: String,
    override val type: String,
    override val href: String,
    val attributes: Attributes,
    val relationships: Relationships? = null
) : IdentityResource

Usage Example ​

kotlin
val literature: Literature = // ... fetched from API

println("${literature.attributes.title}")
println("Type: ${literature.attributes.type.name}") // Manga, Light Novel, etc.

println("Volumes: ${literature.attributes.volumeCount}")
println("Chapters: ${literature.attributes.chapterCount}")
println("Pages: ${literature.attributes.pageCount}")

literature.attributes.stats?.let { stats ->
    println("Rating: ${stats.ratingAverage}/5")
}

// User's reading status
literature.attributes.library?.let { library ->
    println("Status: ${library.status?.name}") // Reading, Completed, etc.
}
